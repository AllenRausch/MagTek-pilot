# 5.2.4 0xD101: Get Property

### Overview

Command 0xD101 retrieves device configuration settings and status information. Properties are identified by Object Identifiers (OIDs) and organized into groups for financial settings, device settings, firmware information, and system status.

**Command ID:** `0xD101`\
**Command Type:** Request/Response (Synchronous)\
**Minimum Firmware:** All versions\
**Device Requirements:** Any DynaFlex family device

### Quick Reference

```python
# Get serial number
property_oid = bytes([0x02, 0x02, 0x01, 0x01, 0x01, 0x01, 0x01])
response = get_property(property_oid)
serial_number = response['value'].decode('ascii')
```

### Command Structure

#### Request Format

```
┌─────────────────────────────────────────────────┐
│ USB HID Report (64 bytes)                       │
├─────────────────────────────────────────────────┤
│ Byte 0: Report ID (0x00)                        │
│ Bytes 1-2: Message Length                       │
│ Byte 3: Message Type (0x02 = Request)           │
│ Bytes 4-5: Command ID (0xD1 0x01)               │
│ Bytes 6-7: Device Family ID                     │
│ Bytes 8+: Property OID(s)                       │
└─────────────────────────────────────────────────┘
```

#### Response Format

```
┌─────────────────────────────────────────────────┐
│ Response Message                                 │
├─────────────────────────────────────────────────┤
│ Byte 0: Report ID (0x00)                        │
│ Bytes 1-2: Message Length                       │
│ Byte 3: Message Type (0x03 = Response)          │
│ Bytes 4-5: Command ID (0xD1 0x01)               │
│ Byte 6: Status Code                             │
│ Bytes 7+: Property Value(s) in TLV format       │
└─────────────────────────────────────────────────┘
```

### Property OID Structure

Properties use Object Identifiers (OIDs) based on ITU-T X.660 standard.

#### OID Format

```
┌──────────────────────────────────────┐
│ OID Octet Encoding                   │
├──────────────────────────────────────┤
│ Example: 1.1.1.1.1.1 (Terminal ID)   │
│                                      │
│ Dot Notation: 1.1.1.1.1.1            │
│ Octet Encoding: 01 01 01 01 01 01    │
└──────────────────────────────────────┘
```

### Single Property Request

#### Example: Get Serial Number

```python
def get_serial_number():
    """Get device serial number"""
    
    # Property OID: 2.2.1.1.1.1.1 (Serial Number)
    oid = bytes([0x02, 0x02, 0x01, 0x01, 0x01, 0x01, 0x01])
    
    # Build request message
    message = bytearray(64)
    message[0] = 0x00  # Report ID
    message[1] = 0x00  # Length MSB
    message[2] = 0x09  # Length LSB (2 family + 7 OID)
    message[3] = 0x02  # Request type
    message[4] = 0xD1  # Command MSB
    message[5] = 0x01  # Command LSB
    message[6] = 0x00  # Device family (auto-detect)
    message[7] = 0x00
    message[8:8+len(oid)] = oid
    
    # Send and receive
    response = send_command(message)
    
    # Parse response
    if response[6] == 0x00:  # Success
        # Skip to TLV data (after status byte)
        tlv_data = response[7:]
        
        # Parse TLV (Tag = OID, Length, Value)
        tag_len = tlv_data[0]
        length = tlv_data[tag_len + 1]
        value = tlv_data[tag_len + 2:tag_len + 2 + length]
        
        return value.decode('ascii')
    
    return None

# Usage
serial = get_serial_number()
print(f"Serial Number: {serial}")
```

#### Example: Get Firmware Version

```python
def get_firmware_version():
    """Get main firmware version"""
    
    # Property OID: 2.1.2.2.2.2 (Main Firmware Version)
    oid = bytes([0x02, 0x01, 0x02, 0x02, 0x02, 0x02])
    
    message = bytearray(64)
    message[0] = 0x00
    message[1] = 0x00
    message[2] = 0x08  # 2 + 6 bytes
    message[3] = 0x02
    message[4] = 0xD1
    message[5] = 0x01
    message[6] = 0x00
    message[7] = 0x00
    message[8:14] = oid
    
    response = send_command(message)
    
    if response[6] == 0x00:
        # Firmware version format: "X.Y.Z.W"
        tlv_data = response[7:]
        tag_len = tlv_data[0]
        length = tlv_data[tag_len + 1]
        value = tlv_data[tag_len + 2:tag_len + 2 + length]
        return value.decode('ascii')
    
    return None

# Usage
version = get_firmware_version()
print(f"Firmware Version: {version}")
```

### Multiple Property Request

You can request multiple properties in a single command:

```python
def get_multiple_properties(oid_list):
    """Get multiple properties at once"""
    
    # Build list of OIDs
    oid_data = bytearray()
    for oid in oid_list:
        oid_data.extend(oid)
    
    # Build message
    message = bytearray(64)
    message[0] = 0x00
    message[1] = 0x00
    message[2] = 2 + len(oid_data)  # Family ID + OIDs
    message[3] = 0x02
    message[4] = 0xD1
    message[5] = 0x01
    message[6] = 0x00  # Device family
    message[7] = 0x00
    message[8:8+len(oid_data)] = oid_data
    
    response = send_command(message)
    
    if response[6] == 0x00:
        return parse_multiple_properties(response[7:])
    
    return None

def parse_multiple_properties(tlv_data):
    """Parse multiple TLV properties"""
    
    properties = []
    i = 0
    
    while i < len(tlv_data):
        # Read tag (OID) length
        tag_len = tlv_data[i]
        if tag_len == 0:
            break
        
        i += 1
        
        # Read tag (OID)
        tag = tlv_data[i:i+tag_len]
        i += tag_len
        
        # Read value length
        value_len = tlv_data[i]
        i += 1
        
        # Read value
        value = tlv_data[i:i+value_len]
        i += value_len
        
        properties.append({
            'oid': tag,
            'value': value
        })
    
    return properties

# Example: Get device info
device_info_oids = [
    bytes([0x02, 0x02, 0x01, 0x01, 0x01, 0x01, 0x01]),  # Serial
    bytes([0x02, 0x01, 0x02, 0x02, 0x02, 0x01]),        # Model
    bytes([0x02, 0x01, 0x02, 0x02, 0x02, 0x02])         # Version
]

properties = get_multiple_properties(device_info_oids)
for prop in properties:
    print(f"OID: {prop['oid'].hex()}")
    print(f"Value: {prop['value']}")
```

### Response Status Codes

| Code   | Meaning            | Action                               |
| ------ | ------------------ | ------------------------------------ |
| `0x00` | Success            | Property value in payload            |
| `0x01` | Invalid parameter  | Check OID format                     |
| `0x02` | Property not found | Verify OID for device model          |
| `0x03` | Access denied      | Property may be read-only or secured |
| `0x10` | Device busy        | Retry after delay                    |

### Error Handling

```python
def safe_get_property(oid, default=None):
    """Get property with error handling"""
    
    try:
        response = get_property(oid)
        
        if response is None:
            print(f"Failed to get property {oid.hex()}")
            return default
        
        status = response.get('status', 0xFF)
        
        error_messages = {
            0x01: "Invalid OID format",
            0x02: "Property not found on this device",
            0x03: "Access denied",
            0x10: "Device busy"
        }
        
        if status != 0x00:
            print(f"Error: {error_messages.get(status, 'Unknown error')}")
            return default
        
        return response['value']
    
    except Exception as e:
        print(f"Exception getting property: {e}")
        return default

# Usage with defaults
battery = safe_get_property(
    bytes([0x02, 0x03, 0x01, 0x02, 0x01, 0x04]),
    default=bytes([0x00])  # Default to 0%
)
```

#### System Health Check

```python
def system_health_check():
    """Check device health status"""
    
    health = {
        'overall': 'OK',
        'issues': []
    }
    
    # Check battery
    battery = safe_get_property(
        bytes([0x02, 0x03, 0x01, 0x02, 0x01, 0x04])
    )
    if battery and battery[0] < 20:
        health['issues'].append(f"Low battery: {battery[0]}%")
    
    # Check tamper
    tamper = safe_get_property(
        bytes([0x02, 0x03, 0x01, 0x01, 0x02, 0x03])
    )
    if tamper and tamper[0] != 0x00:
        health['issues'].append("Tamper detected!")
        health['overall'] = 'CRITICAL'
    
    # Check operational status
    status = safe_get_property(
        bytes([0x02, 0x03, 0x01, 0x02, 0x01, 0x01])
    )
    if status and status[0] != 0x01:
        health['issues'].append("Device not ready")
        health['overall'] = 'WARNING'
    
    # Check keys
    key_status = safe_get_property(
        bytes([0x02, 0x03, 0x01, 0x01, 0x01, 0x01])
    )
    if not key_status or len(key_status) == 0:
        health['issues'].append("No keys injected")
        health['overall'] = 'WARNING'
    
    return health

# Run health check
health = system_health_check()
print(f"\nDevice Health: {health['overall']}")
if health['issues']:
    print("\nIssues:")
    for issue in health['issues']:
        print(f"  - {issue}")
```

### Performance Considerations

#### Batch Requests

When getting multiple properties, use a single request:

```python
# Slow: Multiple requests
serial = get_property(SERIAL_OID)
model = get_property(MODEL_OID)
version = get_property(VERSION_OID)

# Fast: Single request
properties = get_multiple_properties([
    SERIAL_OID,
    MODEL_OID,
    VERSION_OID
])
```

#### Caching

Cache stable properties to reduce USB traffic:

```python
class DeviceProperties:
    def __init__(self):
        self.cache = {}
        self.cache_ttl = {}
    
    def get(self, oid, ttl=None):
        """Get property with optional caching"""
        
        oid_key = oid.hex()
        
        # Check cache
        if ttl and oid_key in self.cache:
            import time
            if time.time() < self.cache_ttl.get(oid_key, 0):
                return self.cache[oid_key]
        
        # Get from device
        value = safe_get_property(oid)
        
        # Cache if TTL specified
        if ttl and value:
            import time
            self.cache[oid_key] = value
            self.cache_ttl[oid_key] = time.time() + ttl
        
        return value

# Usage
props = DeviceProperties()

# Cache serial number for 1 hour (it never changes)
serial = props.get(SERIAL_OID, ttl=3600)

# Always get fresh battery level
battery = props.get(BATTERY_OID)
```
